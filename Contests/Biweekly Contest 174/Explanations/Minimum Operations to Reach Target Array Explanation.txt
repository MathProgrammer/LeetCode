Let us look at all integers i, such that S[i] != T[i] 

Go through each of them one by one and change all the integers to the correct value. 

We only need to know the number of distinct values such that S[i] =/= T[i]

------

class Solution {
public:
    int minOperations(vector<int>& nums, vector<int>& target) 
    {
        set <int> operations; 
        for(int i = 0; i < nums.size(); i++)
        {
            if(nums[i] != target[i])
            {
                operations.insert(nums[i]);
            }
        }
        return operations.size();
    }
};
