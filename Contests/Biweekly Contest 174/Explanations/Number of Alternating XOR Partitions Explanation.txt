Let us first solve it first with only a single target. 

If XOR[L, R] = T, then XOR[1, L - 1] xor XOR[1, R] = T 

So in order to count the number of segments who's XOR is T ending at R, it is
the frequency of XOR[1, r] xor T. 

----- 

How do we count the number of segments ending at i, with T1 ? 

It is the frequency of all segments with 

1. Prefix XOR = Prefix XOR[i] xor T1 
2. Is a valid ending of segment 2. 

It is the similar thing for T2. 

---- 

Maintain the following information 

1. Prefix XOR 
2. good_endings_1[x] -> Number of segments with ending of T1 who's prefix XOR is x 
3. good_endings_2[x] -> Number of segments with ending of T2 who's prefix XOR is x. 
4. answer_1[i] -> Number of segments 1 ending at index i 
5. answer_2[i] -> Number of segments 2 ending at index i 

----

The transition is quite simple 

answer_1[i] = good_endings_2[prefix_xor^T1]
answer_2[i] = good_endings_1[prefix_xor^T2] 

Also update 
good_endings_1[prefix_xor] += good_endings_2[prefix_xor^T]
good_endings_2[prefix_xor] += good_endings_2[prefix_xor^T]

-----

class Solution {
public:
    int alternatingXOR(vector<int>& nums, int target1, int target2) 
    {
        const int MOD = 1e9 + 7;
        vector <long long> answer_1(nums.size()), answer_2(nums.size());
        map <int, long long> good_endings_1, good_endings_2;
        int prefix_xor = 0;
        good_endings_2[prefix_xor] = 1;
        for(int i = 0; i < nums.size(); i++)
        {
            prefix_xor ^= nums[i];
            
            int segment_1_xor = prefix_xor^target1; 
            int good_endings_1_here = good_endings_2[segment_1_xor];

            int segment_2_xor = prefix_xor^target2; 
            int good_endings_2_here = good_endings_1[segment_2_xor];

            good_endings_1[prefix_xor] += good_endings_1_here;
            good_endings_1[prefix_xor] %= MOD;
            answer_1[i] = good_endings_1_here;
            
            good_endings_2[prefix_xor] += good_endings_2_here;
            good_endings_2[prefix_xor] %= MOD;
            answer_2[i] = good_endings_2_here;
        }

        long long answer = answer_1.back() + answer_2.back();
        answer %= MOD;
        return answer;
    }
};
